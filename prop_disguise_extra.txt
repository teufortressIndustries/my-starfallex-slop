--@name Prop Disguise (Stealth + Physics)
--@author You
--@shared

if SERVER then
    local ply = owner()
    local currentModel = "models/maxofs2d/hover_rings.mdl"

    local propHolo = holograms.create(ply:getPos(), ply:getAngles(), currentModel)
    local physProp = nil

    local inPhysicsMode = false
    local wasReloading = false
    local wasRightClick = false
    local wasZooming = false -- Track the Zoom key state
    local isFrozen = false
    local lastJump = 0
    local wasAlive = true 

    local FORCE_MULTIPLIER = 25
    local JUMP_MULTIPLIER = 250

    local holoOffsetZ = 0

    ply:setModelScale(0)

    -- Networking helper to keep the client HUD informed
    local function syncState()
        if not isValid(ply) then return end
        net.start("prop_disguise_sync")
        net.writeBool(inPhysicsMode)
        net.writeBool(isFrozen)
        net.writeString(currentModel)
        net.send(ply)
    end

    -- Uses native engine parenting to eliminate Think loop tracking overhead
    local function updateHologram()
        if not isValid(propHolo) then return end
        
        propHolo:setParent(ply)
        propHolo:setLocalPos(Vector(0, 0, holoOffsetZ))
        propHolo:setLocalAngles(Angle(0, 0, 0))
    end

    -- Fixed typo and added a callback delay to allow engine model bounds to load
    local function updateHoloOffset(callback)
        timer.simple(0.05, function()
            if isValid(propHolo) and propHolo.obbMins then
                local mins = propHolo:obbMins()
                holoOffsetZ = -mins.z
            end
            if callback then callback() end
        end)
    end

    timer.simple(0.1, function()
        updateHoloOffset(function()
            updateHologram()
            syncState()
        end)
    end)

    local function setPlayerHidden(hidden)
        local alpha = hidden and 0 or 255
        local renderMode = hidden and 1 or 0

        ply:setColor(Color(255, 255, 255, alpha))
        ply:setRenderMode(renderMode)
    end

    local function cleanupProp()
        if isValid(physProp) then
            physProp:remove()
        end

        physProp = nil
        isFrozen = false

        if isValid(propHolo) then
            propHolo:setColor(Color(255, 255, 255, 255))
        end
    end

    local function spawnProp()
        if isValid(propHolo) then
            propHolo:setColor(Color(255, 255, 255, 0))
        end

        pcall(function()
            physProp = prop.create(
                ply:getPos() + Vector(0, 0, holoOffsetZ),
                Angle(0, ply:getAngles().y, 0),
                currentModel
            )

            physProp:setOwner(ply)

            timer.simple(0.05, function()
                if not isValid(physProp) then return end
                local phys = physProp:getPhysicsObject()
                if isValid(phys) then
                    phys:wake()
                end
            end)
        end)
    end

    local function handlePhysicsMovement()
        if not isValid(physProp) then return end

        local propPos = physProp:getPos()
        local propRadius = physProp:obbMaxs().y + 16 
        
        local playerFacing = ply:getEyeAngles():getForward()
        playerFacing.z = 0
        playerFacing:normalize()
        local targetPos = propPos - (playerFacing * propRadius)

        local traceData = {
            start = targetPos + Vector(0, 0, 50),
            endpos = targetPos - Vector(0, 0, 200),
            filter = function(ent)
                if ent == ply or ent == physProp then return false end
                return true
            end
        }
        
        local tr = trace.line(traceData.start, traceData.endpos, traceData.filter)
        ply:setPos(tr.Hit and tr.HitPos or targetPos)

        local phys = physProp:getPhysicsObject()
        if not isValid(phys) then return end

        local rightClick = ply:keyDown(IN_KEY.ALT2)

        if rightClick and not wasRightClick then
            isFrozen = not isFrozen
            syncState()

            if isFrozen then
                phys:enableMotion(false)
                print("Prop Frozen")
            else
                phys:enableMotion(true)
                phys:wake()
                print("Prop Unfrozen")
            end
        end

        wasRightClick = rightClick

        if isFrozen then return end

        local ang = ply:getEyeAngles()
        local fwd = ang:getForward()
        local right = ang:getRight()

        fwd.z = 0
        right.z = 0
        fwd:normalize()
        right:normalize()

        local force = Vector(0, 0, 0)

        if ply:keyDown(8) then force = force + fwd end
        if ply:keyDown(16) then force = force - fwd end
        if ply:keyDown(512) then force = force - right end
        if ply:keyDown(1024) then force = force + right end

        if force:getLength() > 0 then
            force:normalize()
            local com = phys:getMassCenter()
            local worldCOM = physProp:localToWorld(com)
            phys:applyForceOffset(force * phys:getMass() * FORCE_MULTIPLIER, worldCOM)
        end

        if ply:keyDown(2) and (timer.curtime() - lastJump > 1) then
            phys:applyForceCenter(Vector(0, 0, phys:getMass() * JUMP_MULTIPLIER))
            lastJump = timer.curtime()
        end
    end

    setPlayerHidden(true)

    hook.add("Think", "PropLogic", function()
        if not isValid(ply) then return end

        local isAlive = ply:isAlive()
        if not isAlive then
            if wasAlive then
                wasAlive = false
                cleanupProp() 
                inPhysicsMode = false
                ply:setModelScale(1)
                setPlayerHidden(false)
                
                if isValid(propHolo) then
                    propHolo:setColor(Color(255, 255, 255, 0))
                end
                syncState()
            end
            return 
        else
            if not wasAlive then
                wasAlive = true
                ply:setModelScale(0)
                setPlayerHidden(true)
                
                if isValid(propHolo) then
                    propHolo:setColor(Color(255, 255, 255, 255))
                    updateHoloOffset(function()
                        updateHologram()
                        print("Disguise Reactivated: Hologram Mode")
                        syncState()
                    end)
                else
                    print("Disguise Reactivated: Hologram Mode")
                    syncState()
                end
            end
        end

        if inPhysicsMode and not isValid(physProp) then
            inPhysicsMode = false
            cleanupProp()
            updateHologram()
            print("Your physical disguise was DESTROYED! Reverting to Hologram Mode.")
            syncState()
        end

        -- Mode Toggling (ALT1)
        local isReloading = ply:keyDown(IN_KEY.ALT1)
        if isReloading and not wasReloading then
            inPhysicsMode = not inPhysicsMode

            if not inPhysicsMode and isValid(physProp) then
                ply:setPos(physProp:getPos() - Vector(0, 0, holoOffsetZ))
            end

            cleanupProp()

            if inPhysicsMode then
                spawnProp()
                print("Switched to: Physics Mode! Use WASD to roll.")
            else
                updateHologram()
                print("Switched to: Hologram Mode")
            end
            syncState()
        end
        wasReloading = isReloading

        -- Copy Looked-at Prop Model (Zoom Key)
        local isZooming = ply:keyDown(IN_KEY.ZOOM)
        if isZooming and not wasZooming then
            local tr = ply:getEyeTrace()
            if isValid(tr.Entity) and not tr.Entity:isPlayer() then
                local modelPath = tr.Entity:getModel()
                if modelPath and modelPath ~= "" and modelPath:match("%.mdl$") then
                    currentModel = modelPath

                    if isValid(propHolo) then
                        propHolo:setModel(currentModel)
                    end

                    updateHoloOffset(function()
                        updateHologram()
                        if inPhysicsMode then
                            cleanupProp()
                            spawnProp()
                        end
                        print("Disguise updated via target: " .. currentModel)
                        syncState()
                    end)
                end
            end
        end
        wasZooming = isZooming

        if inPhysicsMode then
            handlePhysicsMovement()
        end
    end)

    hook.add("PlayerSay", "PropDisguiseChat", function(talker, text, teamChat)
        if talker ~= ply then return end

        if text:sub(1, 6):lower() == "!prop " then
            local modelPath = text:sub(7):trim()

            if modelPath and modelPath ~= "" then
                currentModel = modelPath

                if isValid(propHolo) then
                    propHolo:setModel(currentModel)
                end

                updateHoloOffset(function()
                    updateHologram()
                    if inPhysicsMode then
                        cleanupProp()
                        spawnProp()
                    end
                    print("Disguise model updated to: " .. currentModel)
                    syncState()
                end)
            end
            return "" 
        end
    end)

    hook.add("removed", "CleanupDisguise", function()
        cleanupProp()
        ply:setModelScale(1)
        if not isValid(ply) then return end
        setPlayerHidden(false)
    end)

end

if CLIENT then
    if player() ~= owner() then return end

    local cl_inPhysicsMode = false
    local cl_isFrozen = false
    local cl_currentModel = "models/props_junk/watermelon01.mdl"

    net.receive("prop_disguise_sync", function()
        cl_inPhysicsMode = net.readBool()
        cl_isFrozen = net.readBool()
        cl_currentModel = net.readString()
    end)

    hook.add("CalcView", "PropCamera", function(ply, pos, angles, fov)
        local view = {}
        local maxs = ply:obbMaxs()
        local offset = angles:getForward() * -130 + angles:getUp() * (maxs.z * 0.5)
        
        local traceData = {
            start = pos,
            endpos = pos + offset,
            filter = ply
        }
        local tr = trace.line(traceData.start, traceData.endpos, traceData.filter)
        
        view.origin = tr.HitPos + tr.HitNormal * 4
        view.angles = angles
        view.drawviewer = true
        return view
    end)

    hook.add("draw2d", "PropHUD", function()
        render.setColor(Color(15, 15, 15, 220))
        render.drawRect(30, 30, 250, 110)
        
        render.setFont("DermaDefault")
        
        render.setColor(Color(255, 255, 255))
        render.drawText(40, 40, "=== DISGUISE ACTIVE ===")
        
        if cl_inPhysicsMode then
            render.setColor(Color(255, 140, 0))
            render.drawText(40, 60, "Mode: PHYSICS (WASD to roll)")
            
            if cl_isFrozen then
                render.setColor(Color(255, 60, 60))
                render.drawText(40, 75, "Status: FROZEN [ALT2]")
            else
                render.setColor(Color(60, 255, 60))
                render.drawText(40, 75, "Status: UNANCHORED [ALT2]")
            end
        else
            render.setColor(Color(0, 165, 255))
            render.drawText(40, 60, "Mode: HOLOGRAM (Follows You)")
            render.setColor(Color(180, 180, 180))
            render.drawText(40, 75, "Press ALT1 to manifest physical prop")
        end
        
        render.setColor(Color(200, 255, 200))
        render.drawText(40, 90, "Look at a prop & press ZOOM to copy")

        local cleanModel = cl_currentModel:match("([^/]+)%.mdl$") or cl_currentModel
        render.setColor(Color(255, 255, 255, 140))
        render.drawText(40, 110, "Model: " .. cleanModel)
    end)
end
