--@name Prop Disguise Extra
--@author John Doe
--@shared

--------------------------------------------------------------------------------
--  CONFIGURATION MENU (Adjust settings easily here!)
--------------------------------------------------------------------------------
local ROLL_SPEED      = 25   -- How fast you roll around in regular Omni Mode
local VEHICLE_SPEED   = 35   -- How fast you move forward/backward in Vehicle Mode
local STEERING_SPEED  = 25    -- How fast you turn left/right in Vehicle Mode
local JUMP_HEIGHT     = 250    -- How high you can jump as a prop
local DEFAULT_MODEL   = "models/props_junk/watermelon01.mdl" -- Starting prop

--------------------------------------------------------------------------------
--  CONTROLS CONFIGURATION (Rebind your keys easily here!)
--------------------------------------------------------------------------------
local KEY_UNDO_CYCLE   = IN_KEY.ALT1      -- Toggle Holo Mode / Cycle possessed props (Duck+ALT1 to cycle forward)
local KEY_FREEZE       = IN_KEY.ALT2      -- Freeze / Unfreeze your prop in place
local KEY_TOGGLE_STYLE = IN_KEY.SPEED     -- Change rolling style (Omni vs Vehicle)
local KEY_DETACH       = IN_KEY.WALK      -- Detach from prop / Attach back to it
local KEY_POSSESS      = IN_KEY.RELOAD    -- Crosshair trace possess while detached
local KEY_COPY_PROP    = IN_KEY.ZOOM      -- Copy model of prop you are looking at

if SERVER then
    local ply = owner()
    local currentModel = DEFAULT_MODEL
    
    local currentSkin = 0
    local currentMaterial = ""
    local currentColor = Color(255, 255, 255, 255)
    local currentScale = Vector(1, 1, 1)

    local propHolo = holograms.create(ply:getPos(), ply:getAngles(), currentModel)
    local physProp = nil
    local vehicleChair = nil

    local possessionHistory = {} 
    local historyIndex = 0

    local inPhysicsMode = false
    local isVehicleMode = false 
    local isPossessing = false 
    local isDetached = false 
    local wasReloading = false
    local wasRightClick = false
    local wasWalk = false
    local wasSneak = false 
    local wasZooming = false
    local isFrozen = false
    local lastJump = 0
    local wasAlive = true 

    local undoKeyPressTime = 0
    local holdThreshold = 0.4 

    local holoOffsetZ = 0
    local detachedHoloPos = Vector(0, 0, 0)
    local detachedHoloAng = Angle(0, 0, 0)

    ply:setModelScale(0)

    local function addPropToHistory(ent)
        if not isValid(ent) then return end
        if possessionHistory[historyIndex] == ent then return end
        
        if historyIndex < #possessionHistory then
            for i = #possessionHistory, historyIndex + 1, -1 do
                table.remove(possessionHistory, i)
            end
        end
        
        table.insert(possessionHistory, ent)
        if #possessionHistory > 10 then 
            table.remove(possessionHistory, 1)
        end
        historyIndex = #possessionHistory
    end

    local function applyVisualProperties(ent)
        if not isValid(ent) then return end
        ent:setSkin(currentSkin)
        ent:setMaterial(currentMaterial)
        ent:setRenderMode(1) 
        ent:setColor(Color(currentColor.r, currentColor.g, currentColor.b, currentColor.a))
        
        pcall(function()
            if ent.setMatrix then
                local mat = Matrix()
                mat:scale(currentScale)
                ent:setMatrix(mat)
            elseif ent.setScale then
                ent:setScale(currentScale)
            end
        end)
    end

    local function updateHologram()
        if not isValid(propHolo) then return end
        
        if inPhysicsMode then
            propHolo:setColor(Color(0,0,0,0))
            return
        end

        if isDetached then
            propHolo:setParent(nil)
            propHolo:setPos(detachedHoloPos)
            propHolo:setAngles(detachedHoloAng)
        else
            propHolo:setParent(ply)
            propHolo:setLocalPos(Vector(0, 0, holoOffsetZ))
            propHolo:setLocalAngles(Angle(0, 0, 0))
        end
        applyVisualProperties(propHolo)
    end

    local function updateHoloOffset(callback)
        timer.simple(0.05, function()
            if isValid(propHolo) and propHolo.obbMins then
                local mins = propHolo:obbMins()
                holoOffsetZ = -mins.z * currentScale.z
            end
            if callback then callback() end
        end)
    end

    timer.simple(0.1, function()
        updateHoloOffset(function()
            updateHologram()
            print("[Prop Disguise] Loaded! Use keys or Zoom to copy props.")
        end)
    end)

    local function setPlayerHidden(hidden)
        local alpha = hidden and 0 or 255
        local renderMode = hidden and 1 or 0
        ply:setColor(Color(255, 255, 255, alpha))
        ply:setRenderMode(renderMode)
    end

    local function cleanupProp()
        if isValid(ply) and ply:inVehicle() then
            ply:exitVehicle()
        end
    
        if isValid(vehicleChair) then
            vehicleChair:remove()
            vehicleChair = nil
        end
    
        if isValid(physProp) and not isPossessing then
            physProp:remove()
        end
    
        physProp = nil
        isFrozen = false
        isDetached = false
        isPossessing = false
    
        if isValid(propHolo) then
            propHolo:setColor(currentColor) 
        end
    end

    local function checkAndCreateChair(targetProp)
        if not isValid(targetProp) then return end
        
        if not isValid(vehicleChair) then
            vehicleChair = prop.createSent(targetProp:getPos(), Angle(), "Chair_Plastic", true)
            if isValid(vehicleChair) then
                vehicleChair:setNoDraw(true)
                vehicleChair:setNocollideAll(true)
            end
        end

        if isValid(vehicleChair) and not isDetached and not ply:inVehicle() then
            ply:enterVehicle(vehicleChair)
        end
    end

    local function spawnProp()
        if isValid(propHolo) then
            propHolo:setColor(Color(currentColor.r, currentColor.g, currentColor.b, 0))
        end
    
        physProp = prop.create(
            ply:getPos() + Vector(0, 0, holoOffsetZ),
            Angle(0, ply:getAngles().y, 0),
            currentModel
        )
    
        isPossessing = false
    
        timer.simple(0, function()
            if not isValid(physProp) then return end 
            applyVisualProperties(physProp)
                
            local phys = physProp:getPhysicsObject()
            if isValid(phys) then
                phys:wake()
            end
    
            checkAndCreateChair(physProp)
            print("Physical Prop Disguise Spawned.") 
        end)
    end

    local function handlePhysicsMovement()
        if not isValid(physProp) then return end

        local phys = physProp:getPhysicsObject()
        if not isValid(phys) then return end

        if isValid(vehicleChair) then
            vehicleChair:setPos(physProp:getPos()+Vector(0,0,10))
            vehicleChair:setAngles(Angle(0))
            vehicleChair:setFrozen(true)
        end

        if isDetached then return end

        local rightClick = ply:keyDown(KEY_FREEZE)
        if rightClick and not wasRightClick then
            isFrozen = not isFrozen
            if isFrozen then
                phys:enableMotion(false)
                print("Prop Frozen in place!")
            else
                phys:enableMotion(true)
                phys:wake()
                print("Prop Unfrozen!")
            end
        end
        wasRightClick = rightClick

        if isFrozen then return end

        local com = phys:getMassCenter()
        local worldCOM = physProp:localToWorld(com)

        if isVehicleMode then
            local propForward = physProp:getAngles():getForward()
            local propUp = physProp:getAngles():getUp()
            
            local groundTrace = trace.line(
                worldCOM, 
                worldCOM - Vector(0, 0, (physProp:obbMaxs().z * currentScale.z) + 10), 
                {ply, physProp}
            )
            local isGrounded = groundTrace.Hit

            if isGrounded then
                local throttle = 0
                if ply:keyDown(IN_KEY.FORWARD) then throttle = throttle + 1 end
                if ply:keyDown(IN_KEY.BACK) then throttle = throttle - 1 end

                if throttle ~= 0 then
                    local linearForce = propForward * throttle * phys:getMass() * VEHICLE_SPEED
                    phys:applyForceOffset(linearForce, worldCOM)
                end

                local steering = 0
                if ply:keyDown(IN_KEY.MOVELEFT) then steering = steering + 1 end
                if ply:keyDown(IN_KEY.MOVERIGHT) then steering = steering - 1 end

                if steering ~= 0 then
                    local steerTorque = propUp * steering * phys:getMass() * STEERING_SPEED
                    phys:applyTorque(steerTorque)
                end
            end
        else
            local ang = ply:getEyeAngles()
            local fwd = ang:getForward()
            local right = ang:getRight()

            fwd.z = 0
            right.z = 0
            fwd:normalize()
            right:normalize()

            local force = Vector(0, 0, 0)

            if ply:keyDown(IN_KEY.FORWARD) then force = force + fwd end
            if ply:keyDown(IN_KEY.BACK) then force = force - fwd end
            if ply:keyDown(IN_KEY.MOVELEFT) then force = force - right end
            if ply:keyDown(IN_KEY.MOVERIGHT) then force = force + right end

            if force:getLength() > 0 then
                force:normalize()
                phys:applyForceOffset(force * phys:getMass() * ROLL_SPEED, worldCOM)
            end
        end

        if ply:keyDown(IN_KEY.JUMP) and (timer.curtime() - lastJump > 1) then
            phys:applyForceCenter(Vector(0, 0, phys:getMass() * JUMP_HEIGHT))
            lastJump = timer.curtime()
        end
    end

    local function switchPossessedProp(targetProp)
        physProp = targetProp
        isPossessing = true
        inPhysicsMode = true
        isDetached = false
        
        currentModel = physProp:getModel()
        currentSkin = physProp:getSkin() or 0
        currentMaterial = physProp:getMaterial() or ""
        currentColor = physProp:getColor() or Color(255, 255, 255, 255)
        
        local phys = physProp:getPhysicsObject()
        if isValid(phys) then
            phys:enableMotion(true)
            phys:wake()
        end
        
        if isValid(propHolo) then
            propHolo:setModel(currentModel)
            propHolo:setColor(Color(0, 0, 0, 0))
        end
        
        local propOwnerName = "the Map/Server"
        if physProp.getOwner then
            local propOwner = physProp:getOwner()
            if isValid(propOwner) and propOwner:isPlayer() then
                propOwnerName = propOwner:getName()
            end
        end

        checkAndCreateChair(physProp)
        print("Possessing Prop: " .. currentModel .. " (Owned by: " .. propOwnerName .. ")")
    end

    setPlayerHidden(true)

    hook.add("Think", "PropLogic", function()
        if not isValid(ply) then return end

        local isAlive = ply:isAlive()
        if not isAlive then
            if wasAlive then
                wasAlive = false
                cleanupProp() 
                inPhysicsMode = false
                ply:setModelScale(1)
                setPlayerHidden(false)
                possessionHistory = {} 
                historyIndex = 0
                if isValid(propHolo) then
                    propHolo:setColor(Color(255, 255, 255, 0))
                end
                print("Player died. Disguise system wiped.") 
            end
            return 
        else
            if not wasAlive then
                wasAlive = true
                ply:setModelScale(0)
                setPlayerHidden(true)
                if isValid(propHolo) then
                    propHolo:setColor(currentColor)
                    updateHoloOffset(function()
                        updateHologram()
                        print("Disguise Reactivated: Hologram Mode")
                    end)
                end
            end
        end

        if inPhysicsMode and not isDetached and isValid(vehicleChair) then
            if not ply:inVehicle() then
                ply:enterVehicle(vehicleChair)
            end
        end

        if inPhysicsMode and not isValid(physProp) then
            inPhysicsMode = false
            cleanupProp()
            updateHologram()
            print("Your physical disguise was DESTROYED! Reverting to Hologram Mode.")
        end

        local isReloading = ply:keyDown(KEY_UNDO_CYCLE)
        if isReloading and not wasReloading then
            undoKeyPressTime = timer.curtime()
        elseif not isReloading and wasReloading then
            local holdDuration = timer.curtime() - undoKeyPressTime

            if holdDuration >= holdThreshold then
                inPhysicsMode = not inPhysicsMode
                if not inPhysicsMode and isValid(physProp) then
                    ply:setPos(physProp:getPos() - Vector(0, 0, holoOffsetZ))
                end
                cleanupProp()
                if inPhysicsMode then
                    print("Switched to PHYSICAL mode.") 
                    spawnProp()
                else
                    print("Switched to HOLOGRAM mode.") 
                    updateHologram()
                end
            else
                if inPhysicsMode and isPossessing then
                    local wantRedo = ply:keyDown(IN_KEY.DUCK)
                    local foundValid = false

                    if wantRedo then
                        for i = historyIndex + 1, #possessionHistory do
                            local historicalProp = possessionHistory[i]
                            if isValid(historicalProp) then
                                historyIndex = i
                                switchPossessedProp(historicalProp)
                                updateHoloOffset(function() print("-> Redo Forward to stack idx: " .. i) end)
                                foundValid = true
                                break
                            end
                        end
                    else
                        for i = historyIndex - 1, 1, -1 do
                            local historicalProp = possessionHistory[i]
                            if isValid(historicalProp) then
                                historyIndex = i
                                switchPossessedProp(historicalProp)
                                updateHoloOffset(function() print("<- Undo Backward to stack idx: " .. i) end)
                                foundValid = true
                                break
                            end
                        end
                        
                        if not foundValid then
                            print("Undo: No previous valid props remaining in stack. Hold and release to toggle Holo.")
                        end
                    end
                else
                    inPhysicsMode = not inPhysicsMode
                    if not inPhysicsMode and isValid(physProp) then
                        ply:setPos(physProp:getPos() - Vector(0, 0, holoOffsetZ))
                    end
                    cleanupProp()
                    if inPhysicsMode then 
                        print("Toggled mode: PHYSICAL") 
                        spawnProp() 
                    else 
                        print("Toggled mode: HOLOGRAM") 
                        updateHologram() 
                    end
                end
            end
        end
        wasReloading = isReloading

        if inPhysicsMode then
            local isSneak = ply:keyDown(KEY_TOGGLE_STYLE)
            if isSneak and not wasSneak then
                isVehicleMode = not isVehicleMode
                if isVehicleMode then
                    print("Handling changed: VEHICLE MODE (W/S Forward, A/D Turn)")
                else
                    print("Handling changed: OMNI MODE (Classic Ball Rolling)")
                end
            end
            wasSneak = isSneak
        end

        local isWalk = ply:keyDown(KEY_DETACH)
        if isWalk and not wasWalk then
            isDetached = not isDetached
            if isDetached then
                print("Detached from disguise! You can move freely.")
                if inPhysicsMode then
                    if isValid(physProp) and not isPossessing then
                        isPossessing = true 
                        print("Your original prop has been safely left behind.")
                    end
                else
                    if isValid(propHolo) then
                        detachedHoloPos = propHolo:getPos()
                        detachedHoloAng = propHolo:getAngles()
                        updateHologram()
                    end
                end
            else
                if inPhysicsMode then
                    if isValid(physProp) then
                        print("Attached back to prop!")
                    else
                        inPhysicsMode = false
                        updateHologram()
                        print("Returned to Hologram mode.")
                    end
                else
                    if isValid(propHolo) then
                        ply:setPos(detachedHoloPos - Vector(0, 0, holoOffsetZ))
                        updateHologram()
                    end
                    print("Attached back to hologram!")
                end
            end
        end
        wasWalk = isWalk

        local isRealReload = ply:keyDown(KEY_POSSESS)
        local isZooming = ply:keyDown(KEY_COPY_PROP)

        if isRealReload and isDetached then
            local tr = ply:getEyeTrace()
            local target = tr.Entity
            if isValid(target) and not target:isPlayer() then
                if hasPermission("entities.canTool", target) then
                    local modelPath = target:getModel()
                    if modelPath and modelPath ~= "" and modelPath:match("%.mdl$") then
                        
                        if isValid(vehicleChair) then vehicleChair:remove() vehicleChair = nil end
                        
                        currentModel = modelPath
                        currentSkin = target:getSkin() or 0
                        currentMaterial = target:getMaterial() or ""
                        currentColor = target:getColor() or Color(255, 255, 255, 255)
                        
                        switchPossessedProp(target)
                        addPropToHistory(physProp) 
                    else
                        print("Possession Failed: Target doesn't have a valid .mdl path.") 
                    end
                else
                    print("Possession Failed: No permission to tool target.") 
                end
            end
        end

        if isZooming and not wasZooming then
            local tr = ply:getEyeTrace()
            local target = tr.Entity
            if isValid(target) and not target:isPlayer() then
                local modelPath = target:getModel()
                if modelPath and modelPath ~= "" and modelPath:match("%.mdl$") then
                    currentModel = modelPath
                    currentSkin = target:getSkin() or 0
                    currentMaterial = target:getMaterial() or ""
                    currentColor = target:getColor() or Color(255, 255, 255, 255)
                    
                    if isValid(propHolo) then propHolo:setModel(currentModel) end
                    updateHoloOffset(function()
                        updateHologram()
                        if inPhysicsMode then
                            cleanupProp()
                            spawnProp()
                        end
                        print("Copied prop appearance successfully! (" .. currentModel .. ")")
                    end)
                end
            end
        end
        wasZooming = isZooming

        if inPhysicsMode then
            handlePhysicsMovement()
        end
    end)

    hook.add("PlayerSay", "PropDisguiseChat", function(talker, text, teamChat)
        if talker ~= ply then return end
        if text:sub(1, 6):lower() == "!prop " then
            local modelPath = text:sub(7):trim()
            if modelPath and modelPath ~= "" then
                currentModel = modelPath
                if isValid(propHolo) then propHolo:setModel(currentModel) end
                updateHoloOffset(function()
                    updateHologram()
                    if inPhysicsMode then
                        cleanupProp()
                        spawnProp()
                    end
                    print("Model set via Chat: " .. currentModel) 
                end)
            else
                print("Error: Model path cannot be empty.") 
            end
            return "" 
        end
    end)

    hook.add("removed", "CleanupDisguise", function()
        cleanupProp()
        ply:setModelScale(1)
        setPlayerHidden(false)
        print("[Prop Disguise] Expression removed, clean up complete.") 
    end)
end
