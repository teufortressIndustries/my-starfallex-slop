--@name Who is that?
--@author John Doe
--@shared
--@description A cheap knockoff G-Man sightings mod.

if SERVER then
    -- ============================================================
    --  SERVER RECEIVER
    -- ============================================================
    net.receive("gman_spawned_report", function(len, ply)
        local distance = net.readUInt(16)
        print(string.format("[G-Man Sightings] %s SPOTTED G-Man from %d units away!", ply:getName(), distance))
    end)

else
    -- ============================================================
    --  CLIENT WHITELIST
    -- ============================================================
    local WHITELIST_ON = false

    local WHITELIST = {
        ["STEAM_0:0:123456789"] = true,
        ["STEAM_0:0:000000000"] = true,
    }

    if not WHITELIST[player():getSteamID()] and WHITELIST_ON then return end

    -- ============================================================
    --  CONFIG
    -- ============================================================
    local SPAWN_DELAY_MIN   = 60
    local SPAWN_DELAY_MAX   = 1800
    local SPAWN_RADIUS      = 1000
    local SPAWN_BEHIND_BIAS = 0.6
    local SPOT_ANGLE        = 65
    local VANISH_DELAY      = 0.1
    local VANISH_DELAY_MAX  = 0.5
    local GMAN_MODEL        = "models/gman_high.mdl"
    local GMAN_SCALE        = Vector(1, 1, 1)
    local GMAN_MATERIAL     = "models/debug/debugwhite"
    local GMAN_COLOR        = Color(0, 0, 0)

    local SOUND_SPAWN_ENABLED  = true
    local GMAN_SOUND_SPAWN     = "friends/friend_online.wav"
    local SOUND_VANISH_ENABLED = true
    local GMAN_SOUND_VANISH    = "ambient/levels/canals/windchime5.wav"
    local GMAN_SOUND_LEVEL     = 75
    local GMAN_SOUND_PITCH     = 100
    local GMAN_SOUND_VOLUME    = 1

    -- ============================================================
    --  CROWD BIAS CONFIG
    --  Isolated players are targeted more; crowded players get longer waits.
    --  CROWD_BIAS_RADIUS  : units around the player to count nearby players
    --  CROWD_BIAS_STRENGTH: 0 = no effect, 1 = full effect
    --                       At full strength, being alone can halve the delay
    --                       and being in a full crowd can double it.
    -- ============================================================
    local CROWD_BIAS_ENABLED  = true
    local CROWD_BIAS_RADIUS   = 800
    local CROWD_BIAS_STRENGTH = 1.0

    -- ============================================================
    --  STATE
    -- ============================================================
    local gman        = nil
    local spotted     = false
    local vanishTimer = nil
    local spawnTimer  = nil

    -- ============================================================
    --  HELPERS
    -- ============================================================

    local function isOutOfView(pos, fovBias)
        local ply   = player()
        local eyes  = ply:getEyePos()
        local aim   = ply:getAimVector()
        local toPos = (pos - eyes):getNormalized()
        local dot   = aim:dot(toPos)
        return dot < fovBias
    end

    local function hasLineOfSight(pos)
        local ply  = player()
        local eyes = ply:getEyePos()
        local tr   = trace.line(eyes, pos, ply)
        return tr.HitPos:getDistance(pos) < 32
    end

    local function findSpawnPos()
        local ply     = player()
        local plyPos  = ply:getPos()
        local forward = ply:getForward()

        for _ = 1, 30 do
            local randomYaw = math.random() * 360
            local dir = Vector(math.cos(math.rad(randomYaw)), math.sin(math.rad(randomYaw)), 0)

            local behind = -forward
            behind.z = 0
            behind:normalize()
            dir = (dir * (1 - SPAWN_BEHIND_BIAS) + behind * SPAWN_BEHIND_BIAS):getNormalized()

            local dist   = math.random(150, SPAWN_RADIUS)
            local tryPos = plyPos + dir * dist + Vector(0, 0, 100)

            local floorTr = trace.line(tryPos, tryPos - Vector(0, 0, 300), ply)
            if floorTr.Hit then
                local spawnPos = floorTr.HitPos + Vector(0, 0, 2)
                if isOutOfView(spawnPos, SPAWN_BEHIND_BIAS - 0.2) and hasLineOfSight(spawnPos) then
                    return spawnPos
                end
            end
        end

        return nil
    end

    local function playGmanSound(snd, pos)
        sound.emitSound(snd, pos, GMAN_SOUND_LEVEL, GMAN_SOUND_PITCH, GMAN_SOUND_VOLUME)
    end

    -- Returns a delay multiplier based on how crowded the local player is.
    -- crowdFraction 0 (alone)    multiplier < 1  shorter delay (more sightings)
    -- crowdFraction 1 (crowded)  multiplier > 1  longer delay  (fewer sightings)
    local function getCrowdDelayMultiplier()
        if not CROWD_BIAS_ENABLED then return 1 end
    
        local plyPos  = player():getPos()
        local allPlys = find.allPlayers()
        local others  = #allPlys - 1
    
        if others <= 0 then
            return math.max(0.1, 1 - CROWD_BIAS_STRENGTH * 0.5)
        end
    
        local nearby = 0
        for _, p in ipairs(allPlys) do
            if p ~= player() and p:getPos():getDistance(plyPos) < CROWD_BIAS_RADIUS then
                nearby = nearby + 1
            end
        end
    
        local crowdFraction = nearby / others
    
        local mult = 1 + CROWD_BIAS_STRENGTH * (crowdFraction * 1.5 - 0.5)
        return math.max(0.1, mult)
    end

    -- ============================================================
    --  CORE LOGIC
    -- ============================================================

    local function gmanAlive()
        return gman ~= nil and gman:isValid()
    end

    local function removeGman(playVanishSound)
        if playVanishSound and SOUND_VANISH_ENABLED then
            local pos = gmanAlive() and gman:getPos() or player():getPos()
            playGmanSound(GMAN_SOUND_VANISH, pos)
        end

        if gmanAlive() then gman:remove() end
        gman    = nil
        spotted = false
        if vanishTimer then
            timer.remove(vanishTimer)
            vanishTimer = nil
        end
    end

    local function scheduleNextSighting()
        local baseDelay = math.random(SPAWN_DELAY_MIN, SPAWN_DELAY_MAX)
        local delay     = math.clamp(
            baseDelay * getCrowdDelayMultiplier(),
            SPAWN_DELAY_MIN * 0.25,   -- floor: never faster than 25% of min
            SPAWN_DELAY_MAX * 4       -- ceiling: never slower than 4 max
        )

        spawnTimer = "gman_spawn_" .. tostring(math.random(100000))
        timer.simple(delay, function()
            if not gmanAlive() then
                spawnGman()
            else
                scheduleNextSighting()
            end
        end)
    end

    function spawnGman()
        local pos = findSpawnPos()
        if not pos then
            timer.simple(5, spawnGman)
            return
        end

        local ply      = player()
        local toPlayer = (ply:getEyePos() - pos):getAngle()
        local ang      = Angle(0, toPlayer.y, 0)

        gman = hologram.create(pos, ang, GMAN_MODEL, GMAN_SCALE)
        gman:setMaterial(GMAN_MATERIAL)
        gman:setColor(GMAN_COLOR)

        spotted = false

        if SOUND_SPAWN_ENABLED then
            playGmanSound(GMAN_SOUND_SPAWN, pos)
        end

        scheduleNextSighting()
    end

    -- ============================================================
    --  THINK HOOK
    -- ============================================================
    hook.add("think", "gman_vision_check", function()
        if not gmanAlive() then return end
        if spotted then return end

        local ply     = player()
        local eyes    = ply:getEyePos()
        local aim     = ply:getAimVector()
        local gmanPos = gman:getPos() + Vector(0, 0, 40)
        local toGman  = (gmanPos - eyes):getNormalized()

        local dot       = aim:dot(toGman)
        local angleDiff = math.deg(math.acos(math.clamp(dot, -1, 1)))

        if angleDiff < SPOT_ANGLE then
            spotted     = true
            vanishTimer = "gman_vanish_" .. tostring(math.random(100000))

            local distance    = math.floor(eyes:getDistance(gmanPos))
            local t           = math.clamp(distance / SPAWN_RADIUS, 0, 1)
            local vanishDelay = VANISH_DELAY + (VANISH_DELAY_MAX - VANISH_DELAY) * t

            timer.simple(vanishDelay, function()
                removeGman(true)
            end)

            net.start("gman_spawned_report")
            net.writeUInt(distance, 16)
            net.send()
        end
    end)

    -- ============================================================
    --  BOOT
    -- ============================================================
    timer.simple(5, function()
        scheduleNextSighting()
    end)

    hook.add("removed", "gman_cleanup", function()
        removeGman(false)
        if spawnTimer then timer.remove(spawnTimer) end
    end)
end
